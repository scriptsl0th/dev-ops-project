cat <<EOF > first-pod.yaml
apiVersion: v1
kind: Pod
metadata:
  name: first-pod 
  labels:
    app: mon-application
    environment: development
spec:
  containers:
  - name: conteneur-api
    image: hashicorp/http-echo
    args:
    - "-text=Hello Kubernetes 👋"
    ports:
    - containerPort: 5678
EOF

cat first-pod.yaml

kubectl apply -f first-pod.yaml

kubectl get pods

kubectl describe pod first-pod

kubectl expose pod first-pod --type=NodePort --port=5678

// service

nano first-service.yml

apiVersion: v1
kind: Service
metadata:
  name: first-service
spec:
  selector:
    app: mon-application   # doit correspondre au label du Pod
  ports:
  - protocol: TCP
    port: 5678             # port interne du service
    targetPort: 5678       # port exposé par le Pod
    nodePort: 30080        # port externe fixe (entre 30000–32767)
  type: NodePort

nano first_replica.yml

apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: myapp-rs
spec:
  replicas: 3
  selector:
    matchLabels:
      app: myapp
  template:
    metadata:
      labels:
        app: myapp
    spec:
      containers:
      - name: myapp-container
        image: nginx:latest
        ports:
        - containerPort: 80


kubectl apply -f first_service.yml
kubectl apply -f first_replica.yml

kubectl get pods
kubectl get svc
kubectl get rs
